<?php
/**
 * 구인 목록 (index 중앙 - 채용공고)
 * API: GET /api/v1/recruit/notices/?limit=10
 * 승인된 공고만 반환, 프리미엄 공고 우선 정렬
 */
if (!function_exists('dongta_api_get')) {
    include_once(dirname(dirname(__FILE__)) . '/include/api_helper.php');
}

$items = dongta_api_get('/recruit/notices/', ['limit' => 10], 300);
if (!$items) $items = [];
?>
<table width='380' border='0' cellspacing='0' cellpadding='1'>
<?php if ($items): foreach ($items as $row):
    $is_premium = !empty($row['is_premium']);
    $color      = $is_premium ? '#c00' : '#333';
    $bold       = $is_premium ? 'bold' : 'normal';
    $company    = isset($row['company_name']) ? $row['company_name'] : '';
    $title      = isset($row['title'])        ? $row['title']        : '';
    $emp_type   = isset($row['employment_type']) ? $row['employment_type'] : '';
?>
  <tr>
    <td width='5'>&nbsp;</td>
    <td width='10'>
      <?php if ($is_premium): ?>
        <img src='images/icon_premium.gif' width='10' height='10' title='프리미엄' />
      <?php else: ?>
        <img src='images/jum_gray.gif' width='2' height='2' />
      <?php endif; ?>
    </td>
    <td>
      <a href='/recruit/JobofferList.phtml?cmd=read&data_idx=<?= (int)$row['id'] ?>'
         style='font-size:11px; color:<?= $color ?>; font-weight:<?= $bold ?>;'>
        <?php if ($emp_type): ?>
          <span style='color:#666;'>[<?= htmlspecialchars($emp_type, ENT_QUOTES, 'UTF-8') ?>]</span>
        <?php endif; ?>
        <?= htmlspecialchars(dongta_cut_str($title, 46), ENT_QUOTES, 'UTF-8') ?>
      </a>
    </td>
    <td width='55' style='font-size:10px; color:#666; text-align:right;'>
      <?= htmlspecialchars(dongta_cut_str($company, 16), ENT_QUOTES, 'UTF-8') ?>
    </td>
    <td width='35' style='font-size:10px; color:#999; text-align:right;'>
      <?= dongta_date_label($row['created_at']) ?>
    </td>
    <td width='5'>&nbsp;</td>
  </tr>
<?php endforeach; else: ?>
  <tr>
    <td colspan='6' style='font-size:11px; color:#999; padding:5px;'>
      등록된 구인공고가 없습니다.
    </td>
  </tr>
<?php endif; ?>
  <tr>
    <td colspan='6' height='3'></td>
  </tr>
  <tr>
    <td colspan='6' align='right'>
      <a href='/recruit/JobofferList.phtml?pageflag=offer'
         style='font-size:11px; color:#666;'>더보기 &raquo;</a>
    </td>
  </tr>
</table>
